// Chỉ lưu mã này trong Apps Script; không đưa lên kho GitHub công khai. const SHEET_ID = '1Ov6QMRMwv7ndwGTlQKFObAVNcE_TJFcAhwXDtIcDDFk'; const SHEET_NAME = 'BaiLam'; // Giữ cách tính hiện tại; đổi 0.24 thành 0.25 nếu giáo viên chọn. const SCORE_BY_CORRECT = [0, 0.24, 0.5, 0.75, 1]; const ANSWERS = { 1: [true,true,false,true], 2: [true,false,true,false], 3: [true,false,true,true], 4: [true,false,true,true], 5: [true,true,true,false], 6: [true,false,true,false], 7: [true,false,true,false], 8: [true,false,true,false], 9: [true,false,true,true], 10: [true,false,true,true] }; function jsonResponse(value) { return ContentService.createTextOutput(JSON.stringify(value)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const lock = LockService.getScriptLock(); try { const p = e && e.parameter; if (!p || !p.hoTen || !p.lop || !/^\d+$/.test(p.stt || '') || Number(p.stt) < 1 || Number(p.stt) > 60 || !/^[0-9a-f-]{36}$/i.test(p.submissionId || '')) { throw new Error('Thông tin bài nộp không hợp lệ.'); } const answers = JSON.parse(p.answers); if (!answers || typeof answers !== 'object' || Array.isArray(answers)) { throw new Error('Dữ liệu đáp án không hợp lệ.'); } let score = 0, totalCorrect = 0; const details = []; for (let i = 1; i <= 10; i++) { let correct = 0; const choices = []; ['a','b','c','d'].forEach((key, j) => { const value = answers['q' + i + '_' + key]; if (value !== true && value !== false && value !== null) { throw new Error('Thiếu hoặc sai định dạng đáp án.'); } if (value === ANSWERS[i][j]) { correct++; totalCorrect++; } choices.push(key + ':' + (value === true ? 'Đ' : value === false ? 'S' : '_')); }); score += SCORE_BY_CORRECT[correct]; details.push('C' + i + '[' + choices.join(',') + ']'); } // Cho phép trang cũ tiếp tục nộp; trang mới phải có ảnh chụp đề đầy đủ. const snapshot = p.examSnapshot ? JSON.parse(p.examSnapshot) : null; let layout = '', displayed = '', snapshotJson = ''; if (snapshot) { if (!Array.isArray(snapshot) || snapshot.length !== 10) throw new Error('Đề lưu không hợp lệ.'); const ids = new Set(); const layoutParts = [], displayedParts = []; snapshot.forEach((q, index) => { if (!Number.isInteger(q.questionId) || q.questionId < 1 || q.questionId > 10 || ids.has(q.questionId) || typeof q.header !== 'string' || !Array.isArray(q.items) || q.items.length !== 4) throw new Error('Sai thứ tự đề.'); ids.add(q.questionId); const keys = new Set(); const mapping = [], choices = []; q.items.forEach((item, j) => { if (!item || !['a','b','c','d'].includes(item.key) || keys.has(item.key) || typeof item.text !== 'string') throw new Error('Sai thứ tự ý.'); keys.add(item.key); const label = 'abcd'[j]; const value = answers['q' + q.questionId + '_' + item.key]; mapping.push(label + '→' + item.key); choices.push(label + ':' + (value === true ? 'Đ' : value === false ? 'S' : '_')); }); layoutParts.push('Câu ' + (index+1) + '→C' + q.questionId + '[' + mapping.join(',') + ']'); displayedParts.push('Câu ' + (index+1) + '[' + choices.join(',') + ']'); }); layout = layoutParts.join(' | '); displayed = displayedParts.join(' | '); snapshotJson = JSON.stringify({version: p.examVersion || '', questions: snapshot}); if (snapshotJson.length > 45000) throw new Error('Nội dung đề vượt giới hạn lưu.'); } const monitor = p.monitoring ? JSON.parse(p.monitoring) : null; if (monitor && (!Number.isSafeInteger(monitor.hiddenCount) || monitor.hiddenCount < 0 || !Number.isSafeInteger(monitor.fullscreenExitCount) || monitor.fullscreenExitCount < 0 || !Array.isArray(monitor.events) || monitor.events.length > 200 || monitor.events.some(event => !event || !['page_hidden','fullscreen_exit','fullscreen_unsupported','fullscreen_unavailable'].includes(event.type) || typeof event.time !== 'string' || !Number.isFinite(Date.parse(event.time))))) { throw new Error('Dữ liệu giám sát không hợp lệ.'); } const monitorJson = monitor ? JSON.stringify(monitor.events) : ''; if (monitorJson.length > 45000) throw new Error('Nhật ký vượt giới hạn.'); const gradingSnapshot = JSON.stringify({answers: ANSWERS, scoreByCorrect: SCORE_BY_CORRECT}); lock.waitLock(30000); const book = SpreadsheetApp.openById(SHEET_ID); const sheet = book.getSheetByName(SHEET_NAME) || book.insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow(['Thời gian nộp','Họ và tên','Lớp','STT','Điểm','Số ý đúng','Chi tiết theo mã gốc','Mã bài']); } // Chỉ thêm tiêu đề cột mới; giữ nguyên dữ liệu và cột A:H hiện có. sheet.getRange(1,9,1,4).setValues([['Thứ tự đề hiển thị','Lựa chọn theo đề hiển thị','Nội dung đề lúc nộp (JSON)','Đáp án và thang điểm lúc chấm (JSON)']]); sheet.getRange(1,13,1,3).setValues([['Số lần trang bị ẩn','Số lần thoát toàn màn hình','Nhật ký giám sát (JSON)']]); if (sheet.getLastRow() > 1) { const existing = sheet.getRange(2,8,sheet.getLastRow()-1,1) .createTextFinder(p.submissionId).matchEntireCell(true).findNext(); if (existing) return jsonResponse({ok:true,submissionId:p.submissionId,duplicate:true}); } // Giữ chuỗi người dùng dưới dạng văn bản, tránh bị diễn giải thành công thức. const text = value => /^[=+@-]/.test(value.trim()) ? "'" + value.trim() : value.trim(); sheet.appendRow([new Date(),text(p.hoTen),text(p.lop),Number(p.stt), Number(score.toFixed(2)),totalCorrect + '/40',details.join(' | '),p.submissionId,layout,displayed,snapshotJson,gradingSnapshot,monitor ? monitor.hiddenCount : '',monitor ? monitor.fullscreenExitCount : '',monitorJson]); SpreadsheetApp.flush(); return jsonResponse({ok:true,submissionId:p.submissionId}); } catch (err) { console.error(err); return jsonResponse({ok:false,error:'Không lưu được bài. Hãy kiểm tra cấu hình hoặc thử lại.'}); } finally { if (lock.hasLock()) lock.releaseLock(); } }
